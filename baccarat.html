dealtCards.push(b3); bS = (bS + gV(b3)) % 10; } } else if (bS <= 5 && pS >= 6 && pS < 8 && cards.length > 0) { b3 = cards.pop(); dealtCards.push(b3); bS = (bS + gV(b3)) % 10; } // Update shoe in sheet const handsDealt = parseInt(shoeData[shoeRowIdx - 1][4] || 0) + 1;
